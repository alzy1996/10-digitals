# Dallni v0.4 "Real Mall" Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the default 3D look a photo-real mall: CC0 textures mapped in the shader, real HDRI reflections,
furnished shops, backlit fascias and a lamp-lit Walk view ceiling. Showcase gets bloom; weak phones fall back to the
v0.3 toy look with no download.

**Architecture:**
- A fourth scene style, `real`, built on the toy geometry (bevels + normals) plus a per-vertex `surface` id and extra
  procedural props.
- `effectiveStyle(style, quality)` maps `real` + `light` to `toy`, so the auto-downgrade only swaps materials and
  palette.
- A `MeshStandardMaterial` extended with `onBeforeCompile` samples marble, plaster or wood textures by surface id and
  world position.
- A small post module (composer + bloom + Neutral tone mapping) renders `real` at Showcase.
- Assets are same-origin WebP + a downsampled RGBE HDRI.

**Tech Stack:** three.js 0.186.1 (`HDRLoader`, `EffectComposer`, `RenderPass`, `UnrealBloomPass`, `OutputPass`),
troika-three-text 0.52.5, React 18.3.1, Vite 8, Vitest 5, Python 3.13 + Pillow (asset build only).

**Spec:** `mall-navigation/docs/superpowers/specs/2026-10-11-real-mall-design.md`

## Global Constraints

- No new npm dependencies. No CDN or third-party fetch at runtime: assets are served from `public/assets/real/`
  (same origin).
- Meaning colours stay exact: route, rings, markers, pins, sign accent lines, labels, bay paint. Neutral tone mapping
  only; the route teal pixel must match toy within ±6 per channel.
- Light quality = the v0.3 toy look, and requests no real asset.
- Idle = 0 frames. Shadows are redrawn only on scene changes.
- Budgets:
  - `scene3d` ≤ 72 KiB;
  - `engine3d` ≤ 220 KiB;
  - lazy total ≤ 260 KiB → raise to **280 KiB** if needed (report);
  - first screen ≤ 110 KiB;
  - real assets ≤ 3 MB hard (1.5 MB target).
- Honesty: SIMULATION labels unchanged; no people, avatars or parked cars.
- `?style=toy|storm|classic` keep working; the default becomes `real`.
- 8 GB RAM laptop: one headless browser at a time. Close every server at the end (kill orphan node PIDs by port).

```js
function webpSize(buf) {

}
```

```python
"""Build the v0.4 real-mall assets once (outputs are committed; app builds need no Python).
CC0 sources: Poly Haven (marble_tiles, painted_plaster_wall, brown_planks_03, leadenhall_market). Usage:
  python tools/build_real_assets.py            # downloads to .cache/real-src, writes public/assets/real/
"""
import json, math, os, subprocess, sys
from PIL import Image

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.dirname(HERE)
SRC = os.path.join(ROOT, '.cache', 'real-src')
OUT = os.path.join(ROOT, 'public', 'assets', 'real')
MAPS = {
    'marble.webp': ('marble_tiles', 'Diffuse', 80), 'marble-n.webp': ('marble_tiles', 'nor_gl', 85),
    'marble-r.webp': ('marble_tiles', 'Rough', 85), 'plaster.webp': ('painted_plaster_wall', 'Diffuse', 80),
    'plaster-n.webp': ('painted_plaster_wall', 'nor_gl', 85), 'wood.webp': ('brown_planks_03', 'Diffuse', 80),
}

def curl_json(url):
    return json.loads(subprocess.run(['curl', '-s', '-m', '60', url], capture_output=True, text=True, check=True).stdout)

def fetch(asset, kind, ext='jpg', res='1k'):
    files = curl_json('https://api.polyhaven.com/files/' + asset)
    url = files['hdri'][res][ext]['url'] if kind == 'hdri' else files[kind][res][ext]['url']
    dst = os.path.join(SRC, os.path.basename(url))
    if not os.path.exists(dst):
        subprocess.run(['curl', '-s', '-m', '120', '-o', dst, url], check=True)
    return dst

def read_rgbe(path):
    with open(path, 'rb') as f:
        data = f.read()
    i = data.index(b'\n\n') + 2
    j = data.index(b'\n', i)
    dims = data[i:j].decode().split()  # -Y H +X W
    h, w = int(dims[1]), int(dims[3])
    pos, rows = j + 1, []
    for _ in range(h):
        if data[pos] == 2 and data[pos + 1] == 2:
            pos += 4
            chans = []
            for _c in range(4):
                out = bytearray()
                while len(out) < w:
                    n = data[pos]; pos += 1
                    if n > 128:
                        out += bytes([data[pos]]) * (n - 128); pos += 1
                    else:
                        out += data[pos:pos + n]; pos += n
                chans.append(out)
            rows.append([(chans[0][x], chans[1][x], chans[2][x], chans[3][x]) for x in range(w)])
        else:
            rows.append([tuple(data[pos + 4 * x: pos + 4 * x + 4]) for x in range(w)]); pos += 4 * w
    return w, h, rows

def rgbe_to_f(p):
    if p[3] == 0:
        return (0.0, 0.0, 0.0)
    f = math.ldexp(1.0, p[3] - 136)
    return (p[0] * f, p[1] * f, p[2] * f)

def f_to_rgbe(c):
    v = max(c)
    if v < 1e-32:
        return bytes((0, 0, 0, 0))
    m, e = math.frexp(v)
    s = m * 256.0 / v
    return bytes((int(c[0] * s), int(c[1] * s), int(c[2] * s), e + 128))

def downsample_hdr(src, dst):
    w, h, rows = read_rgbe(src)
    W, H = 512, 256
    fx, fy = w // W, h // H
    out = bytearray()
    for y in range(H):
        for x in range(W):
            acc = [0.0, 0.0, 0.0]
            for dy in range(fy):
                for dx in range(fx):
                    c = rgbe_to_f(rows[y * fy + dy][x * fx + dx])
                    acc = [acc[k] + c[k] for k in range(3)]
            out += f_to_rgbe([a / (fx * fy) for a in acc])
    with open(dst, 'wb') as f:
        f.write(b'#?RADIANCE\nFORMAT=32-bit_rle_rgbe\n\n-Y 256 +X 512\n')
        f.write(bytes(out))

def main():
    os.makedirs(SRC, exist_ok=True)
    os.makedirs(OUT, exist_ok=True)
    for name, (asset, kind, q) in MAPS.items():
        Image.open(fetch(asset, kind)).convert('RGB').save(os.path.join(OUT, name), 'WEBP', quality=q, method=6)
    downsample_hdr(fetch('leadenhall_market', 'hdri', 'hdr'), os.path.join(OUT, 'arcade.hdr'))
    total = sum(os.path.getsize(os.path.join(OUT, f)) for f in os.listdir(OUT))
    print('real assets:', round(total / 1024), 'KiB')

if __name__ == '__main__':
    sys.exit(main())
```

```js
 * Furnished shop interiors for the v0.4 real style (DESIGN-REAL.md 4): procedural props by occupant category,
 * merged into the level's opaque mesh (lit, textured by surface) and its exact mesh (glowing screens and menu boards).

import { distToSegment, pointInPolygon, polygonBounds, polygonCentroid } from './geometry.js'

/** Grid spots inside a unit: >= 0.9 m from its walls, >= 1.5 m from its doors. */
export function propSpots(poly, openings, step = 1.6) {
  const b = polygonBounds(poly)
  const out = []
  const edgeD = (p) => Math.min(...poly.map((q, i) => distToSegment(p, q, poly[(i + 1) % poly.length])))
  for (let x = b.minX + 0.9; x <= b.maxX - 0.9 + 1e-9; x += step) {
    for (let y = b.minY + 0.9; y <= b.maxY - 0.9 + 1e-9; y += step) {
      const p = [x, y]
      if (!pointInPolygon(p, poly) || edgeD(p) < 0.9) continue
      if (openings.some((o) => distToSegment(p, o.a, o.b) < 1.5)) continue
      out.push(p)
    }
  }
  return out
}
```

```js
 * v0.4 "real" material (DESIGN-REAL.md 3): one MeshStandardMaterial (vertex colours) whose albedo, normal and
 * roughness come from CC0 photo textures chosen per vertex by `surface` and mapped by world position (floors xz,
 * walls along the dominant horizontal axis), so MeshBuilder needs no UVs and any venue works.

export const SURFACE_PARAMS = Object.freeze([[0.6, 0], [0.35, 0], [0.85, 0], [0.6, 0], [0.5, 0], [0.3, 1], [0.04, 0.1]])

export function createRealMaterial() {
  const mat = new MeshStandardMaterial({ vertexColors: true, roughness: 0.6, metalness: 0 })
  const uniforms = { uMarble: { value: null }, uMarbleN: { value: null }, uMarbleR: { value: null }, uPlaster: { value: null }, uPlasterN: { value: null }, uWood: { value: null }, uHasMaps: { value: 0 } }
  mat.userData.real = uniforms
  mat.customProgramCacheKey = () => 'dallni-real'
  mat.onBeforeCompile = (sh) => {
    Object.assign(sh.uniforms, uniforms)
    sh.vertexShader = 'attribute float surface;\nvarying float vSurf;\nvarying vec3 vWPr;\nvarying vec3 vWNr;\n' + sh.vertexShader.replace(
      '#include <worldpos_vertex>',
      '#include <worldpos_vertex>\nvSurf = surface;\nvWPr = (modelMatrix * vec4(transformed, 1.0)).xyz;\nvWNr = normalize(mat3(modelMatrix) * objectNormal);'
    )
    const params = SURFACE_PARAMS.map(([r, m]) => `vec2(${r.toFixed(2)}, ${m.toFixed(2)})`).join(', ')
    sh.fragmentShader = `uniform sampler2D uMarble, uMarbleN, uMarbleR, uPlaster, uPlasterN, uWood;
uniform float uHasMaps;
varying float vSurf;
varying vec3 vWPr;
varying vec3 vWNr;
const vec2 SP[7] = vec2[7](${params});
vec2 realUv(float rep) {
  vec3 n = abs(vWNr);
  vec2 uv = n.y > 0.7 ? vWPr.xz : (n.x > n.z ? vWPr.zy : vWPr.xy);
  return uv / rep;
}
vec3 realSample(int s) {
  if (uHasMaps < 0.5) return vec3(1.0);
  if (s == 1) return texture2D(uMarble, realUv(4.0)).rgb;
  if (s == 2) return texture2D(uPlaster, realUv(2.5)).rgb * 1.08;
  if (s == 3) return texture2D(uWood, realUv(3.0)).rgb * 1.15;
  if (s == 4) { vec3 c = texture2D(uPlaster, realUv(0.6)).rgb; return mix(vec3(dot(c, vec3(0.333))), c, 0.4) * 1.1; }
  return vec3(1.0);
}
` + sh.fragmentShader
      .replace('#include <map_fragment>', '#include <map_fragment>\nint rs = int(vSurf + 0.5);\ndiffuseColor.rgb *= realSample(rs);')
      .replace('#include <roughnessmap_fragment>', 'float roughnessFactor = SP[rs].x;\nif (rs == 1 && uHasMaps > 0.5) roughnessFactor = mix(0.15, 0.6, texture2D(uMarbleR, realUv(4.0)).g);')
      .replace('#include <metalnessmap_fragment>', 'float metalnessFactor = SP[rs].y;')
      .replace('#include <normal_fragment_maps>', `#include <normal_fragment_maps>
if (uHasMaps > 0.5 && (rs == 1 || rs == 2)) {
  vec3 tn = (rs == 1 ? texture2D(uMarbleN, realUv(4.0)).xyz : texture2D(uPlasterN, realUv(2.5)).xyz) * 2.0 - 1.0;

  vec3 t = normalize(abs(wn.y) > 0.7 ? vec3(1.0, 0.0, 0.0) : cross(vec3(0.0, 1.0, 0.0), wn));

  vec3 pn = normalize(t * tn.x * 0.6 + b * tn.y * 0.6 + wn * tn.z);
  normal = normalize((viewMatrix * vec4(pn, 0.0)).xyz);
}`)
  }
  return mat

export function setRealMaps(mat, maps) {
  const u = mat.userData.real
  u.uMarble.value = maps.marble; u.uMarbleN.value = maps.marbleN; u.uMarbleR.value = maps.marbleR
  u.uPlaster.value = maps.plaster; u.uPlasterN.value = maps.plasterN; u.uWood.value = maps.wood
  u.uHasMaps.value = maps.marble && maps.plaster && maps.wood ? 1 : 0
}

/** Same-origin assets; each failure -> null (one console warning); all failed -> null. */
export async function loadRealAssets(base = 'assets/real/', { timeoutMs = 20000 } = {}) {
  const tl = new TextureLoader()
  const tex = (name, srgb) => new Promise((res) => {
    const timer = setTimeout(() => res(null), timeoutMs)
    try {
      tl.load(base + name, (t) => {
        clearTimeout(timer)
        t.wrapS = t.wrapT = RepeatWrapping
        t.minFilter = LinearMipmapLinearFilter; t.magFilter = LinearFilter; t.anisotropy = 4
        if (srgb) t.colorSpace = SRGBColorSpace
        res(t)
      }, undefined, () => { clearTimeout(timer); res(null) })
    } catch { clearTimeout(timer); res(null) }
  })
  const env = new Promise((res) => {
    const timer = setTimeout(() => res(null), timeoutMs)
    new HDRLoader().load(base + 'arcade.hdr', (t) => { clearTimeout(timer); t.mapping = EquirectangularReflectionMapping; res(t) }, undefined, () => { clearTimeout(timer); res(null) })
  }).catch(() => null)
  const [marble, marbleN, marbleR, plaster, plasterN, wood, envTex] = await Promise.all([
    tex('marble.webp', true), tex('marble-n.webp', false), tex('marble-r.webp', false),
    tex('plaster.webp', true), tex('plaster-n.webp', false), tex('wood.webp', true), env,
  ])
  const all = { marble, marbleN, marbleR, plaster, plasterN, wood, env: envTex }
  if (Object.values(all).every((v) => !v)) { console.warn('Dallni: real-mall assets could not load; plain look'); return null }
  if (Object.values(all).some((v) => !v)) console.warn('Dallni: some real-mall assets are missing')
  return all
}
```

```js
import { HalfFloatType, NeutralToneMapping, Vector2, WebGLRenderTarget } from 'three'
import { EffectComposer } from 'three/addons/postprocessing/EffectComposer.js'
import { RenderPass } from 'three/addons/postprocessing/RenderPass.js'
import { UnrealBloomPass } from 'three/addons/postprocessing/UnrealBloomPass.js'
import { OutputPass } from 'three/addons/postprocessing/OutputPass.js'

export function createPost(renderer) {
  const rt = new WebGLRenderTarget(4, 4, { type: HalfFloatType, samples: 4 })
  const composer = new EffectComposer(renderer, rt)
  const scenePass = new RenderPass(null, null)
  const markerPass = new RenderPass(null, null)
  markerPass.clear = false
  markerPass.clearDepth = true
  const bloom = new UnrealBloomPass(new Vector2(256, 256), 0.35, 0.4, 1.25)
  composer.addPass(scenePass); composer.addPass(markerPass); composer.addPass(bloom); composer.addPass(new OutputPass())
  return {
    render(scene, markerScene, camera) {
      const tm = renderer.toneMapping
      renderer.toneMapping = NeutralToneMapping
      scenePass.scene = scene; scenePass.camera = camera
      markerPass.scene = markerScene; markerPass.camera = camera
      composer.render()
      renderer.toneMapping = tm
    },
    setSize(w, h, dpr) { composer.setPixelRatio(dpr); composer.setSize(w, h); bloom.resolution.set(w / 2, h / 2) },
    dispose() { composer.dispose(); rt.dispose() },
  }
}
```
